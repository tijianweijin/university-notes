\section{信号与系统}

\subsection{信号}

\subsubsection{基本概念}

\begin{enumerate}
  
  \item 概念辨析
  \begin{itemize}
    \item 消息：信号承载的具体内容；
    \item 信息：信息是消息中蕴含的有用内容；
    \item 信号：消息（或信息）的物理表现形式。
  \end{itemize}

  \item 表示：类似函数，有解析式法、图像法、表格法等。

  \item 特征
  \begin{enumerate}[label=(\arabic*)]

    \item 功率
信号归一化瞬时功率可以表示为其Euclid内积的平方，即$p = \left | x(t) \right |^2$。
故有归一化平均功率（即一般所说的功率）。
    \item 能量：将瞬时归一化功率对时间积分即得到信号的能量。

\begin{center}
\begin{tabularx}{0.98\textwidth}{>{\bfseries}p{1.5cm}X X}
\toprule
信号 & 平均归一化功率 & 归一化能量 \\
\midrule
连续信号 & $P =\displaystyle \lim_{a \to \infty} \frac{1}{2a} \displaystyle\int_{-a}^{a} \left | x \left (t \right ) \right |^2 \dd t$ & $E = \displaystyle \lim_{a \to \infty} \displaystyle \int_{-a}^{a} \left | x \left (t \right ) \right |^2 \dd t$ \\
离散信号 & $P =\displaystyle \lim_{N \to \infty} \frac{1}{2N} \displaystyle\sum_{k=-N}^{N} \left | x \left [ k \right ] \right |^2$ & $E = \displaystyle \lim_{N \to \infty} \displaystyle \sum_{k=-N}^{N} \left | x \left [ k \right ] \right |^2$ \\
\bottomrule
\end{tabularx}
\end{center}

  \end{enumerate}
\end{enumerate}

\subsubsection{分类}

\begin{enumerate}
  
  \item 确定信号与随机信号

能给出确定时间函数的信号称为确定信号，否则称为随机信号。

  \item 连续时间信号与离散时间信号
  
连续时间信号对应带有有限个第一类间断点的连续函数，简称连续信号，一般用$x(t)$表示。
连续信号的时间是连续的，但信号取值可以是连续的也可以是离散的。

离散时间信号对应定义域离散的函数，简称离散信号，一般用$x[n]$表示。
离散信号的时间是离散的，但信号取值可以是连续的也可以是离散的。

\begin{methodbox}[区分连续信号与离散信号]
连续信号与离散信号的区别只在于时间是否连续，而不在于幅值是否连续。
\end{methodbox}

  \item 周期信号与非周期信号

  连续周期信号可以写成$x(t)=x(t+nT)$，其中$T$是周期信号的周期。

  离散周期信号有$x[n]=x[n+N]$，记$\min N$为其基本周期。

  \item 能量受限信号与功率受限信号

能量受限信号$ \Leftrightarrow 0 < E < \infty, P = 0 $ \\
功率受限信号$ \Leftrightarrow 0 < P < \infty, E = \infty $ 
\end{enumerate}

\subsubsection{经典信号}

注：如未说明，默认$t \in \left ( -\infty, \infty \right )$。

\begin{enumerate}
    \item 连续时间信号
    \begin{itemize}
      \item 正弦信号：$x(t)=A\sin\left ( \varOmega t + \varphi \right )$
      
      $A$为幅值，控制信号幅度；$\varOmega$为角频率，控制信号周期；$\varphi$为相位，控制信号相位。

      \item 指数信号：$x(t)=Ae^{st}, \, s = \sigma + \jj \varOmega$
    
      $\sigma$控制指数增大/减小速率，$\varOmega$控制振荡周期。

      \begin{methodbox}[正弦信号与指数信号关系]
        \[
        \sin \left ( \varOmega t \right ) = \frac{1}{\jj 2} \left ( e^{\jj \varOmega t} - e^{-\jj \varOmega t} \right ), \qquad
        \cos \left ( \varOmega t \right ) = \frac{1}{2} \left ( e^{\jj \varOmega t} + e^{-\jj \varOmega t} \right )
        \]
      \end{methodbox}
    
      \item 取样信号（Sa信号）：$\operatorname{Sa}(t)=\dfrac{\sin t}{t}, \, \displaystyle x(t) = A \, \mathrm{Sa}(\varOmega t) = A \, \frac{\sin \varOmega t}{\varOmega t}$
      
      \begin{methodbox}[sinc函数]
        当 $A=1, \, \varOmega = \ppi$ 时，$\mathrm{Sa}(\varOmega t)$ 称为 sinc 函数，
        记为 $\mathrm{sinc}(t)$。
        \[
        \mathrm{sinc}(t)=
        \begin{cases}
        \dfrac{\sin(\ppi t)}{\ppi t}, & t\neq 0,\\[6pt]
        1, & t=0.
        \end{cases}
        \]
      \end{methodbox}
    \end{itemize}

    \item 奇异信号
    
    \begin{itemize}
      \item 单位斜变信号
      \[
      r(t)=
        \begin{cases}
        0, & t < 0,\\
        t, & t \geq 0.
        \end{cases}
      \]
      \item 单位阶跃信号
      \[
      u(t)=
        \begin{cases}
        0, & t < 0,\\
        1, & t \geq 0.
        \end{cases}
      \]
      \item 单位冲激信号（Dirac定义）
      \[
      \int_{-\infty}^{\infty} \delta(t) \dd t = 1, \qquad \delta(t) = 0, \, \forall \, t \neq 0
      \]

      \begin{methodbox}[【技巧】单位冲激信号用处]
        \begin{itemize}
          \item 取样：$\displaystyle x(\tau) = \int_{-\infty}^{\infty} x(t) \delta(t-\tau) \dd t$
        \end{itemize}
      \end{methodbox}

      \begin{methodbox}[【技巧】三奇异信号关系]
        后者是前者的导数，前者是后者的积分。
        \[
        r(t) \to u(t) \to \delta(t)
        \]
        \[
        \dfrac{\dd r(t)}{\dd t} = u(t), \qquad \dfrac{\dd u(t)}{\dd t} = \delta(t)
        \]
      \end{methodbox}

      \begin{methodbox}[矩形信号]
        持续时间为$[\tau_1, \tau_2]$的矩形信号可表示为两阶跃信号的差，即
        \[
        x(t) = A \left [ u(t-\tau_1) - u(t-\tau_2) \right ]
        \]
      \end{methodbox}

      \begin{methodbox}[符号信号]
        符号函数记为
        \[
          \operatorname{sgn}(t) =
          \begin{cases}
            1, & t > 0,\\
            -1, & t < 0.
          \end{cases}
          =2 u(t)-1
        \]
      \end{methodbox}

    \end{itemize}

    \item 其他特殊信号

    \begin{itemize}
      \item 三角信号
      
      最大值为$E$，长度为$T$的三角信号可以视为两个值为$\sqrt{E}$，长度为$\dfrac{T}{2}$的矩形信号的差。
      
      其Fourier变换为
      \[
      \FF \left\{ x(t) \right\} = \dfrac{ET}{2} \Sa^2 \left( \frac{T}{4} \varOmega \right)
      \]
    \end{itemize}

  \end{enumerate}
  \subsubsection{运算}

\begin{enumerate}

  \item 几何变换
    
  几何变换可以用通式$\displaystyle y(t)=x\left [ a\left ( t - \dfrac{b}{a} \right ) \right ]$表示。
    
    \begin{enumerate}[label=(\arabic*)]
      \item 移位：$y(t)=x(t-\tau)$
      \item 反褶：$y(t)=x(-t)$
      \item 尺度变换：$y(t)=x(at)$
    \end{enumerate}
  
    \begin{methodbox}[【技巧】几何变换顺序]
      一般将信号函数整理为$\displaystyle y(t)=x\left [ a\left ( t - \dfrac{b}{a} \right ) \right ]$
      的形式后再考虑如何变换，否则容易出现问题。
    \end{methodbox}

  \item 微积分
  
    \begin{enumerate}[label=(\arabic*)]
      \item 微分：$y(t)=\dfrac{\dd x(t)}{\dd t}$
      \item 积分：$y(t)=\displaystyle \int_{-\infty}^t x(\tau) \dd \tau$
    \end{enumerate}

  \item 复合运算
    \begin{enumerate}[label=(\arabic*)]
      \item 加法：$y(t)=x_1(t) + x_2(t)$
      \item 乘法：$y(t)=x_1(t) \cdot x_2(t)$
      \item 卷积：$y(t)=x(t) * h(t) = \displaystyle \int_{-\infty}^{\infty} x(\tau) h(t-\tau) \dd \tau$
    \end{enumerate}
    \begin{methodbox}[卷积意义]
      \begin{itemize}
        \item 面积解释： 当两个信号均为非负信号时，卷积值可理解为两信号乘积的重叠“面积”；若信号存在正负，则应理解为乘积的代数积分，而不能简单看成几何面积。

        \item 系统意义： 卷积本质上就是 LTI 系统对各时刻输入所产生响应的叠加。
      \end{itemize}
    \end{methodbox}
    \begin{methodbox}[卷积性质]
      \begin{enumerate}

      \item \textbf{代数性质}

      \begin{itemize}
        \item 交换律
        \[
        x_1(t)\ast x_2(t)
        =
        x_2(t)\ast x_1(t)
        \]

        \item 分配律
        \[
        x_1(t)\ast \left[x_2(t)+x_3(t)\right]
        =
        x_1(t)\ast x_2(t)
        +
        x_1(t)\ast x_3(t)
        \]

        \item 结合律
        \[
        \left[x_1(t)\ast x_2(t)\right]\ast x_3(t)
        =
        x_1(t)\ast\left[x_2(t)\ast x_3(t)\right]
        \]

      \end{itemize}

      \item \textbf{与奇异信号的卷积}

      \begin{itemize}
        \item 与单位冲激卷积
        \[
        x(t)\ast\delta(t)=x(t)
        \]

        即 $\delta(t)$ 为卷积运算的恒等元。

        \item 与移位冲激卷积
        \[
        x(t)\ast\delta(t-t_0)
        =
        x(t-t_0)
        \]

        即与移位冲激卷积可使信号整体时移 $t_0$。

        \item 与冲激导数卷积
        \[
        x(t)\ast\delta'(t)
        =
        x'(t)
        \]

        即与 $\delta'(t)$ 卷积相当于对信号求导。
      \end{itemize}

      \item \textbf{微分与积分性质}

      \begin{itemize}
        \item 微分性质
        \[
        \left[x_1(t)\ast x_2(t)\right]'
        =
        x_1'(t)\ast x_2(t)
        =
        x_1(t)\ast x_2'(t)
        \]

        \item 积分性质
        \[
        x_1^{(-1)}(t)\ast x_2(t)
        =
        x_1(t)\ast x_2^{(-1)}(t)
        \]

        其中
        \[
        x^{(-1)}(t)
        =
        \int_{-\infty}^{t}x(\tau)\dd\tau.
        \]

  \end{itemize}
  若卷积的某一函数求导完是单位冲激函数，常常一个求导一个积分，使计算简洁，如、
  \[
  y(t) = x(t) \ast u(t) = \int_{-\infty}^{t}x(\tau)\dd\tau \ast \delta(t) = \int_{-\infty}^{t}x(\tau)\dd\tau
  \]
\end{enumerate}
    \end{methodbox}
    \begin{methodbox}[【技巧】图解法]
      \begin{enumerate}[label=(\arabic*)]
        \item 反褶： 将 \(h(\tau)\) 关于纵轴反褶，得到\(h(-\tau)\)
      
        \item 平移： 将 \(h(-\tau)\) 沿 \(\tau\) 轴平移 \(t\)，得到\(h(t-\tau)\)
        
        \item 相乘： 将 \(x(\tau)\) 与 \(h(t-\tau)\) 相乘，得到 \(y(t)\)
        
        \item 积分： 对整个 \(\tau\) 轴积分
        \[
        y(t)
        =
        \int_{-\infty}^{+\infty}
        x(\tau)h(t-\tau)\dd\tau.
        \]
      \end{enumerate}
    \end{methodbox}
    
\end{enumerate}

\subsubsection{分量与分解}

  信号分解为正交分量时，各分量间不相互干扰，且各分量的能量之和等于原信号的能量。
  若任意两信号的Euclid内积为零，则称这两个信号在时域中正交，即
  \[
  \left \langle  x_n, x_m \right \rangle  = \int_{-\infty}^{\infty} x_n(t) \cdot x_m(t) \dd t = 0, \, \forall \, t \in \left ( -\infty, \infty\right )
  \]
  若额外满足$\displaystyle \left \langle  x_n, x_n \right \rangle  = \left | x_n \right | ^2 = 1$
  则称之为单位正交函数。

  下面介绍几种常见的正交分量。

\begin{enumerate}

  \item 直流分量与交流分量
  对于周期为$T$的周期信号$x(t)$，其交直流分量分别为
  
  \[
  x_\mathrm{DC}(t) = \frac{1}{T} \int_{t_0}^{t_0+T} x(t) \dd t
  \]
  \[
  x_\mathrm{AC}(t) = x(t) - x_\mathrm{DC}(t)
  \]

  直流分量平均量等于原信号平均量，交流分量平均量为零。二者有功率关系
  \[
  P=\left | x_\mathrm{DC}(t) \right |^2 + \frac{1}{T} \int_{t_0}^{t_0+T} \left | x_\mathrm{AC}(t) \right |^2 \dd t
  \]

  \item 奇分量与偶分量
  
  任意信号都可以分解为奇分量与偶分量，分解式如下
  \[
  x_\mathrm{e} = \frac{x(t) + x(-t)}{2}
  ,\qquad
  x_\mathrm{o} = \frac{x(t) - x(-t)}{2}
  \]

  \item 实分量与虚分量
  
  若信号为复信号，则可以分解为实分量与虚分量，分解式如下
  \[
  x_\mathrm{R}(t)=\dfrac{x(t)+x^\ast(t)}{2}
  ,\qquad
  x_\mathrm{I}(t)=\dfrac{x(t)-x^\ast(t)}{\jj 2}
  \]

\end{enumerate}

\subsection{系统}

\subsubsection{表示}

\begin{itemize}
  \item 基本运算单元
  \item 线性常系数微分方程
\end{itemize}

\subsubsection{基本概念}
系统是对输入信号进行加工、变换、处理，并产生输出信号的物理装置。

\subsubsection{冲激响应与阶跃响应}

冲激响应是系统对单位冲激信号的响应，记为$h(t)$。冲激响应是系统的特征函数，能够完全描述系统的性质。

阶跃响应是系统对单位阶跃信号的响应，记为$g(t)$。

\begin{methodbox}[二者关系]
  \[
  g(t)=\int_{-\infty}^{t}h(\tau)\dd\tau=h(t) \ast u(t),\, h(t)=\frac{\dd g(t)}{\dd t}
  \]
\end{methodbox}

任意输入信号$x(t)$的输出信号$y(t)$可以表示为冲激响应的卷积，即
\[
y(t) = x(t) \ast h(t)
\]

\subsubsection{典型冲激响应}

\begin{center}
\begin{tabularx}{0.98\textwidth}{>{\bfseries}p{2cm}X}
\toprule
系统类型 & 冲激响应 $h(t)$ \\
\midrule
        标量乘法器 & $h(t)=a\delta(t)$ \\
        微分器 & $h(t)=\dfrac{\mathrm{d}\delta(t)}{\mathrm{d}t}=\delta'(t)$ \\
        积分器 & $h(t)=\displaystyle\int_{-\infty}^{t}\delta(\tau)\,\mathrm{d}\tau=u(t)$ \\
        延时器 & $h(t)=\delta(t-\tau)$ \\
\bottomrule
\end{tabularx}
\end{center}

\subsubsection{分类}

\begin{enumerate}
  
  \item 线性系统与非线性系统
  
  线性系统满足齐次性和叠加性，亦即映射$\mathcal{T} : x(t) \mapsto y(t)$是线性算子。
  
  \begin{methodbox}[线性系统的判定]
  
    \[\mathcal{T}\left\{ax_1(t) + bx_2(t)\right\} = a\mathcal{T}\left\{x_1(t)\right\} + b\mathcal{T}\left\{x_2(t)\right\}\]
  
  一般线性系统的响应可以写成各输入信号的线性组合。

  加法、微分、积分等运算都是线性算子。
  
  注意：表达式带常数的系统一般不是线性系统，例如$y(t)=2x(t)+3$。
  可用$\mathcal{T}\left\{x(0)\right\}=0$快速判断。
  
  \end{methodbox}

  \item 时变系统与时不变系统
  
  时不变系统的运行规则不会随时间推移发生改变，亦即映射$\mathcal{T} : x(t) \mapsto y(t)$不是$t$的函数。

  \begin{methodbox}[时不变系统的判定]
    \[\mathcal{T}\left\{x(t-t_0)\right\} = y(t-t_0)\]
    
    常见反例：$y(t)=x(2t)$，$y(t)=tx(t)$。
  \end{methodbox}

  \item 因果系统与非因果系统
  
  因果系统的输出仅由当前和过去的输入决定，亦即$y(t_0)$只依赖于$x(t), \, t \in \left ( -\infty, t_0 \right ]$。

  \begin{methodbox}[因果系统的判定]
    因果系统冲激响应满足$h(t)=0,\, t<0$的一般都是非因果系统。
    
    一般包含$x(t+a),\, a>0$的都是非因果系统，如$y(t)=x(t+1)$。
  \end{methodbox}

  \item 稳定系统与不稳定系统

  稳定系统（BIBO稳定）在接入有限大小的输入信号时，能够产生有限范围的输出信号。

  \begin{methodbox}[稳定系统的判定]
    \[\int_{-\infty}^{+\infty} h(t) \dd t < \infty\]
    
    常见反例：$h(t)=sin(\omega_0 t)u(t)$。
  \end{methodbox}

  \begin{methodbox}[因果稳定系统的判定]

\subsection{频域分析}

\begin{methodbox}[时频不确定性原理]

  $\Delta t$与$\Delta f$分别表示以能量分布定义的标准差宽度，则有
  \[
    \Delta t \cdot \Delta f \geq \frac{1}{4\ppi}
  \]

  或写成角频率版本则为
  \[
    \Delta t \cdot \Delta \varOmega \geq \frac12
  \]

  在时谱与频谱中的关系体现为时谱越宽，频谱越窄，反之亦然。
  这是由Fourier变换性质决定的。比较常见的例子是
  \[
  x(t)=\delta (t) \leftrightarrow X(\varOmega)=1, \qquad x(t)=1 \leftrightarrow X(\varOmega)=2\ppi \delta (\varOmega)
  \]
  \[
  x(t)=\operatorname{rect}(t) \leftrightarrow X(\varOmega)=\operatorname{Sa}\left (\dfrac{\varOmega}{2}\right ) , \qquad x(t)= \operatorname{Sa}(t) \leftrightarrow X(\varOmega)=\ppi \operatorname{rect}(\dfrac{\varOmega}{2})
  \]

\end{methodbox}

\begin{methodbox}[频域分析的其他零极点形式]
  零点矢量可以写作$\jj\varOmega-z_j=N_j e^{\jj\psi_j}$，极点矢量可以写作$\jj\varOmega-p_i=M_i e^{\jj\theta_i}$。

\end{methodbox}

\subsection{无失真传输}

  \]
  \item 频域条件：
  \[
  \left|H(\jj\varOmega)\right|=K, \qquad \varphi(\varOmega)=\varOmega t_0
  \]
\end{enumerate}

\subsection{理想模拟低通滤波器}

\begin{enumerate}
  \item 响应函数特征：

  \begin{itemize}
    \item 时域响应：$h(t)=K\Sa(t-t_0)$
    \item 幅频响应：$\left|H(\jj\varOmega)\right|=K\left(u(\varOmega_c)-u(-\varOmega_c)\right)$
    \item 相频响应：$\varphi(\varOmega)=\varOmega t_0$
  \end{itemize}

\begin{methodbox}[【理解】理想模拟低通滤波器不存在]
  理想模拟低通滤波器是非因果系统（冲激响应为取样函数），现实中无法制造。
\end{methodbox}

  \item 系统函数特征

  \begin{methodbox}[【题型】由平方响应函数求系统函数]
    \begin{enumerate}[label=(\arabic*)]
      \item 作代换$s=\jj\varOmega$
      \item 求零点与极点
      \item 取左半平面极点写出$H(s)$
      
      例：$\left|H(\jj\varOmega)\right|^2=\dfrac{4(\varOmega^4+64)}{\varOmega^4-10\varOmega^2+169}$
      
      解：
      \[
      H(s)H(-s)=\left.\frac{4(\varOmega^4+64)}{\varOmega^4-10\varOmega^2+169}\right|_{\varOmega=-\jj s}=\frac{4(s^4+64)}{s^4+10s^2+169}
      \]
      
      解得零点$s_\text{z}=\pm2\pm\jj 2$，极点$s_\text{p}=\pm2\pm\jj 2$。
      选取左半平面极点$s_\text{p}=-2\pm\jj 2$，则
      \[
      H(s)=H_0\frac{s^2+4s+8}{s^2+4s+13}
      \]

      又根据$\left.H(s)\right|_{s=0}=\left.H(\jj\varOmega)\right|_{\varOmega=0}$，
      得$H_0=2$，则

      \[
      H(s)=2\frac{s^2+4s+8}{s^2+4s+13}
      \]
    \end{enumerate}
  \end{methodbox}
\end{enumerate}

\subsubsection{Butterworth滤波器}

\begin{methodbox}[设计Butterworth滤波器]
  
  需要参数：$A_p$、$\varOmega_p$、
  $A_s$、$\varOmega_s$、$\varOmega_c$
  
  有如下关系：
  \[
  A_p=10\lg\left(\frac{\varOmega_p}{\varOmega_c}\right)
  \]
  \[
  A_s=10\lg\left(\frac{\varOmega_s}{\varOmega_c}\right)
  \]

  \begin{enumerate}
    \item 确定滤波器阶数$N$
    \[
    N=\left\lceil\frac{\lg{\dfrac{10^{0.1A_p}-1}{10^{0.1A_s}-1}}}{\lg {\dfrac{\varOmega_p}{\varOmega_s}}}\right\rceil
    \]

    \item 求滤波器极点
    \[
    p_k=\exp\left[\displaystyle\jj\dfrac{\uppi}{2}+\displaystyle\jj\dfrac{(2k+1)\uppi}{2N}\right], \quad k=0,\,1,\,\ldots,\,2N-1
    \]

    \item 求滤波器传递函数
    
    取左半平面极点，即$k=0,\,1,\,\ldots,\,N-1$
    \[
    H(s)=\frac{\varOmega_c^N}{\displaystyle\prod_{k=0}^{N-1}(s-p_k\varOmega_c)}
    \]

    也可查表求传递函数分母，常见分母如下
    \begin{itemize}
      \item $N$=1：$p+1$
      \item $N$=2：$p^2+\sqrt{2}p+1$
      \item $N$=3：$(p^2+p+1)(p+1)$
    \end{itemize}
  \end{enumerate}
\end{methodbox}
