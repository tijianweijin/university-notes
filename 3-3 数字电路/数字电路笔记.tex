\subsection{数制}

\begin{enumerate}
  \item 不同的数制
  \begin{center}
    \begin{tabularx}{0.92\textwidth}{>{\bfseries}p{2.5cm}X X}
      \toprule
      数制（符号） & 基数 & 数码 \\
      \midrule
      二进制 B & 2 & 0,1 \\
      八进制 O & 8 & 0-7 \\
      十进制 D & 10 & 0-9 \\
      十六进制 H & 16 & 0-9, A-F \\
      \bottomrule
    \end{tabularx}
  \end{center}

  \item 数制转换
\begin{methodbox}[非十进制互转]
每3位或4位分组，然后展开或缩写即可。例：
\[
\left ( 11001101.011 \right )_2 = \left ( 315.3 \right )_8 = \left ( CD.6 \right )_{16}
\]
\end{methodbox}

\begin{methodbox}[非十进制转十进制]
按权展开即可。例：
\[
\left ( 101.01 \right )_2 = 1 \times 2^2 + 0 \times 2^1 + 1 \times 2^0 + 0 \times 2^{-1} + 1 \times 2^{-2} = (5.25)_{10}
\]
\[
\left ( 73.4 \right )_8 = 7 \times 8^1 + 3 \times 8^0 + 4 \times 8^0 + 4 \times 8^{-1} = (59.5)_{10}
\]
\[
\left ( 3\mathrm{D}.\mathrm{C} \right )_{16} = 3 \times 16^1 + 13 \times 16^0 + 12 \times 16^{-1} = (61.75)_{10}
\]
\end{methodbox}

\begin{methodbox}[十进制转非十进制]
整数部分除基取余，从低到高；小数部分乘基取整，从高到低。

例：13.25

13除以2余1，6除以2余0，3除以2余1，1除以2余1，整数部分为1101；

0.25乘以2得0.5，取整为0，0.5乘以2得1.0，取整为1，小数部分为01；

故$(13.25)_{10} = \left ( 1101.01 \right )_2$
\end{methodbox}
\end{enumerate}

\subsection{编码}

\begin{enumerate}
  \item 原码、反码与补码
  \begin{center}
    \begin{tabularx}{0.92\textwidth}{>{\bfseries}p{1.5cm} >{}p{9.5cm} X}
      \toprule
      编码类型 & 描述 & 缺点 \\
      \midrule
      原码 & 首位表示符号位，剩余位为二进制绝对值 & 减法运算困难 \\
      反码 & 正数与零不变；负数的符号位不变，其余位为原码取反 & 有两个反码表示零 \\
      补码 & 正数与零不变；负数的补码为反码加1 & 无 \\
      \bottomrule
    \end{tabularx}
  \end{center}

  \begin{methodbox}[补码取负]
    对补码取负数的方法：对补码再取一次补码。
  \end{methodbox}

  \item 二进制编码
  \textbf{二进制数码不一定表示二进制数}。

  最常见的二进制编码是自然二进制码，即将其他数制的数转为二进制后的结果，不赘述。

  \item 二-十进制编码（BCD码）
  \begin{center}
    \begin{tabularx}{0.92\textwidth}{>{\bfseries}p{2.5cm}X X}
      \toprule
      编码类型 & 权 & 特性 \\
      \midrule
      8421码 & 8, 4, 2, 1 & 简便直观 \\
      2421码 & 2, 4, 2, 1 & 对9自补码 \\
      余3码 & 无（由8421码+3得到） & 对9自补码 \\
      5421码 & 5, 4, 2, 1 & 可以产生对称方波输出 \\
      Gray码 & 无 & 相邻两数码只有一位二进制数不同 \\

\subsubsection{逻辑代数基本规则}

    \begin{enumerate}
      \item 代入规则：用逻辑表达式$F$代入某个变量$X$，新逻辑表达式仍然成立。
      \item 反演规则

      将原函数$F$作如下代换：

      \begin{itemize}
        \item $ + \leftrightarrow \cdot $
        \item $ 1 \leftrightarrow 0 $
        \item $ A \leftrightarrow \overline{A} $
      \end{itemize}

      即得到反函数$\overline{F}$。反函数输出的值\textbf{与原函数相反}。

      \item 对偶规则

      将原函数$F$作如下代换：

      \begin{itemize}
        \item $ + \leftrightarrow \cdot $
        \item $ 1 \leftrightarrow 0 $
      \end{itemize}

      即得到对偶函数$F'$。对偶函数\textbf{不需要代换反变量}。

      若$F$与$G$相等，则$F'$与$G'$也相等。

    \end{enumerate}

  \item 代数法
    根据逻辑代数公式，对逻辑函数进行化简。常见方法有：
    \begin{itemize}
      \item 并项法：$AB+A\overline{B}=A$

      \]
      \item 消元法：$A+\overline{A}B=A+B$
    \end{itemize}

  \item Karnaugh图法

\begin{methodbox}[Karnaugh图的数学结构]

\begin{enumerate}
    \item \textbf{输入空间：}
    对于 $n$ 个逻辑变量，所有输入组合构成二进制空间
    \[
        \{0,1\}^n,
    \]
    可视为 $n$ 维超立方体 $Q_n$ 的 $2^n$ 个顶点。
    两顶点 $x,y$ 的 Hamming 距离为
    \[
        d_H(x,y)=\sum_{i=1}^n |x_i-y_i|.
    \]
    当 $d_H(x,y)=1$ 时，两输入组合仅有一个变量不同，对应超立方体中的一条边。

    \item \textbf{Gray坐标：}
    Karnaugh图按 Gray 码排列各坐标，使相邻码字满足
    \[
        d_H=1.
    \]
    因而Karnaugh图中的几何邻接对应逻辑空间中的边邻接，即
    \[
        \boxed{\text{几何相邻}
        \Longleftrightarrow
        \text{仅一个逻辑变量发生变化}}.
    \]
    图的上下、左右边界同样相邻，因此Karnaugh图具有周期边界。

    \item \textbf{Descartes积分解：}
    将 $n$ 个变量分为两组，分别作为行、列坐标，则
    \[
        Q_n \cong Q_p \square Q_q,
        \qquad p+q=n,
    \]
    其中 $\square$ 表示图的笛卡尔积。
    例如四变量Karnaugh图满足
    \[
        Q_4\cong Q_2\square Q_2,
    \]
    行、列均按
    \[
        00,\ 01,\ 11,\ 10
    \]
    的 Gray 顺序排列，从而将四维二进制空间表示为二维坐标表。

    \item \textbf{子立方体与矩形：}
    一个逻辑乘积项中，未出现的变量可以自由取 $0$ 或 $1$。
    若有 $k$ 个自由变量，则该项对应一个含
    \[
        2^k
    \]
    个顶点的 $k$ 维子立方体。

    若其中 $r$ 个自由变量属于行坐标，其余 $k-r$ 个属于列坐标，则该子立方体在Karnaugh图中表现为
    \[
        2^r\times 2^{k-r}
    \]
    的矩形，因此其面积恒为
    \[
        \boxed{2^k}.
    \]

    \item \textbf{代数消元：}
    相邻顶点仅在一个变量上互补，因此可利用
    \[
        AX+A\overline{X}
        =A(X+\overline{X})
        =A
    \]
    消去该变量。
    每使一个变量自由变化，相当于沿一个坐标方向扩展一维，因此
    \[
        \boxed{
        2^k\text{ 个相邻格组成的子立方体}
        \Longleftrightarrow
        \text{消去 }k\text{ 个变量}
        }.
    \]

    例如
    \[
        AB\overline{C}+ABC
        =AB(\overline{C}+C)
        =AB,
    \]
    几何上对应两个顶点沿 $C$ 方向合并为一条边。

    \item \textbf{与或式的几何意义：}
    一个乘积项对应 $\{0,1\}^n$ 中一个轴对齐的子立方体；
    逻辑或对应这些子立方体的集合并。因此
    \[
        F=P_1+P_2+\cdots+P_m
    \]
    可理解为用若干子立方体覆盖
    \[
        F^{-1}(1)
        =\{x\in\{0,1\}^n\mid F(x)=1\}.
    \]
    Karnaugh图化简的本质即为：
    \[
        \boxed{
        \text{用尽可能大的子立方体覆盖全部 }1\text{ 格}
        }.
    \]
    对偶地，或与式则通过圈取 $0$ 格，用子立方体描述
    $F^{-1}(0)$。
\end{enumerate}

\[
\boxed{
\begin{array}{c|c}
\text{逻辑代数} & \text{离散几何} \\ \hline
n\text{ 个变量} & n\text{ 维超立方体 }Q_n\\
\text{一个输入组合} & \text{一个顶点}\\
d_H=1 & \text{一条边}\\
\text{最小项} & \text{一个顶点}\\
\text{一个变量自由} & \text{沿一维扩展}\\
2^k\text{ 项合并} & k\text{ 维子立方体}\\
\text{消去 }k\text{ 个变量} & \text{增加 }k\text{ 个自由维度}\\
\text{逻辑或} & \text{集合并}\\
\text{与或式化简} & \text{用大子立方体覆盖 }F^{-1}(1)
\end{array}}
\]

\[
\boxed{
\text{Karnaugh图}
=
\text{以Gray码保持邻接关系的超立方体二维坐标化}
}
\]

\end{methodbox}

\end{enumerate}

\subsection{小规模组合电路（Small Scale Integrated Circuites, SSI）}

小规模组合电路是指集成电路中逻辑门的数量不超过10个的组合逻辑电路。

小规模组合电路的表达式可以整理为与非·与非式或或非·或或式，从而实现为双轨输入电路。

\begin{enumerate}
  \item 双轨输入电路


\begin{itemize}
  \item 与非·与非式

可以运用反演律将最简与或式化为与非·与非式，用两级与非门实现。

例：$F=A\overline{B}+AC=\overline{(\overline{A\overline{B}})\cdot(\overline{AC})}$

  \item 或非·或非式

可以运用反演律将最简或与式化为或非·或非式，用两级或非门实现。

例：$F=(A+\overline{B})(A+C)=\overline{(\overline{A+\overline{B}})+(\overline{A+C})}$


\end{itemize}

与非·与非式和或非·或非式各有优势，不存在绝对的优劣。
判断哪种方式能最大程度简化时，应根据具体逻辑函数设计比较。

  \item 单轨输入电路

单轨输入电路可以通过给每个输入加非门实现为双轨输入电路，不多叙述。

\end{enumerate}

\subsubsection{逻辑冒险}

\begin{methodbox}[逻辑冒险的判断与消除方法]
  \begin{enumerate}[label=\arabic*.]
    \item 代数式法
    
    只有同时存在正反变量（如$A$与$\overline{A}$）的逻辑函数才可能出现逻辑冒险。

    将标准与或式/或与式中该正反变量以外的所有变量设为$0$或$1$，将逻辑函数化为
    \[
    F=A+\overline{A}
    \]
    的形式，此时存在偏0型冒险；或化为
    \[
    F=A\cdot\overline{A}
    \]
    的形式，此时存在偏1型冒险。

    若要消除逻辑冒险，则在标准与或式/或与式中增加冗余项即可。荣誉项确定办法为
    将存在逻辑冒险时的所有变量组合为与式或或式，作为冗余项加入原逻辑函数中。

    例：$B=0,\,C=1$时存在偏0型冒险，冗余项为$+\overline{B}C$；
    $B=0,\,C=1$时存在偏1型冒险，冗余项为$\cdot(\overline{B}+C)$。

    \item Karnaugh图法

    判断Karnaugh图中是否存在相切圈，若存在则说明存在逻辑冒险现象。

    画圈使相切圈相接，即可消除逻辑冒险现象。
  \end{enumerate}

\end{methodbox}

\subsubsection{功能冒险}

在组合电路中，当有\textbf{两个或两个以上输入变量同时}发生变化，变化前后电路的输出相同，而在
输入变量发生变化时可能出现瞬时错误输出，这种现象称为静态功能冒险

由状态A到状态B的过程中，若首尾电平一致，中间出现不一样的电平，则称为功能冒险。
功能冒险的决定因素如下：
\begin{enumerate}[label=(\arabic*)]
  \item 变量变化两个及以上；
  \item 首尾电平状态一致（从0到0，从1到1）；
  \item 中间电平变化（从0到1，从1到0）。
\end{enumerate}

消除功能冒险的方法包括：
\begin{itemize}
  \item 添加滤波电容，但会破坏输出波形；
  \item 加选通脉冲，对\textbf{逻辑冒险和功能冒险都有效}。
\end{itemize}

\begin{methodbox}
  \textbf{注意}：添加冗余项只能消除逻辑冒险而非功能冒险！
\end{methodbox}

逻辑冒险与功能冒险合称静态冒险。

\subsubsection{动态冒险}

在输入变化前、后函数值不同；且在输入信号变化瞬间，输出将变化三次或更高的奇数次。此处不讨论。

\subsection{多输出函数}
\begin{enumerate}
  \item 用卡诺图分别对每个函数进行化简，并用箭头连线表示出所有的公用圈；
  
  \item 从各个函数\textbf{相同最小项}出发，试图改变原来圈法，以求得更多的公用圈。

  \textbf{原则：}

  \begin{itemize}
    \item 若改圈法后能使总圈数减少（指不同圈），则改；
    \item 若使总圈数增加，则不改；
    \item 若总圈数不变，则取最大的合并圈，使变量输入端减少；
    \item 这一工作可重复进行多次。\textbf{注意：单个变量的圈不用修改}
  \end{itemize}

  \item 写出多输出函数的表达式，并绘出逻辑图。
\end{enumerate}

\subsection{中规模组合电路（Medium Scale Integrated Circuites, MSI）}

\subsubsection{数码比较器}

\begin{enumerate}
  \item TTL中规模集成4位数码比较器74LS85
  
  直观理解即可，比较两个4位二进制数的大小关系，输出$A>B$、$A=B$、$A<B$；
  额外设$G'$（大于）、$E'$（等于）、$S'$（小于）输入端，表示两数相等时的输出。
  
  Verilog HDL实现代码如下：

  \begin{lstlisting}[language=Verilog]
  module compare_4(Y, A, B);
      input [3:0] A, B;
      output [2:0] Y;
      reg [2:0] Y;

      always @(A or B)
      begin
          if (A > B)
              Y = 3'b100;
          else if (A == B)
              Y = 3'b010;
          else
              Y = 3'b001;
      end
  endmodule
  \end{lstlisting}

  功能扩展：

  \begin{itemize}
    \item 5位比较器
    
    使用一片元件，将两个5位二进制数的高4位输入到比较器中，低1位输入到比较器的$A>B$、$A<B$端口中；
    不可以比较相等数值。

    \item 串行级联比较器
    
    将多片比较器的输出端口与$G'$、$E'$、$S'$端口连接起来，即可实现串行级联比较器。

    注意：前后顺序为\textbf{先低后高}，即第一级电路比较最低4位；
    第一级电路应将$G'$、$S'$接地而$E'$接高电平，保证第一级电路正确输出相等结果。
    
    \item 并行级联比较器
    
    基于5位比较器修改得来。使用5片4位比较器并联作为一级电路，1片4位比较器作为二级电路。

    除最低4位接$E'=1$保证相等输出外，一级电路比较器输入方法如5位比较器，最多可以比较24位二进制数；
    二级电路输入法也同5位比较器，一级电路的$A>B$端口接到二级电路的$a_i$端口，$A<B$端口接到二级电路的$b_i$端口，
    最低四位输出接到$G'$、$E'$、$S'$端口。
  \end{itemize}
\end{enumerate}

\subsubsection{编码器与优先编码器}

\begin{enumerate}
  \item 二-十位编码器C304
  
  真值表即8421BCD码。
  \[
  \begin{aligned}
  D &= 8+9\\
  C &= 4+5+6+7\\
  B &= 2+3+6+7\\
  A &= 1+3+5+7+9
  \end{aligned}
  \]

  \item 8线-3线编码器（略）
  \item 优先编码器74148
  
  功能描述：
  
  输入端口$I_0\sim I_7$，使能端$\overline{ST}$；
  输出端口$Y_0\sim Y_2$，使能输出端$Y_{\mathrm{S}}$，
  扩展输出端$\overline{Y_{\mathrm{EX}}}$。

  若使能端为1，全部输出1，代表停止输出；
  
  若使能端为0，输入端全为1，输出端除$Y_{\mathrm{S}}$是0外，其余均为1，
  代表没有输出信号；

  若使能端为0，输入端有信号，则输出端$Y_{\mathrm{S}}$为1，
  输出端$Y_0\sim Y_2$为输入端中最高位的二进制编码。

  简记为：非编码状态时$\overline{Y_{\mathrm{EX}}}$为1；
  编码状态且无输入信号时$Y_{\mathrm{S}}$为0；
  其他状态编码最高位。

  功能扩展：
  \begin{itemize}
    \item 16线至4线的优先编码器
    
    2片74148串联，第一片编码8~15，第二片编码0~7；第一片的输出端接到第二片的使能端。

    输出端为\textbf{反码}$F_0\sim F_3$，扩展端$F_{\mathrm{EX}}$。

    片1的$\overline{Y_{\mathrm{EX}}}$作为4位码最高位$F_3$输出，
    两片对应位$Y_2\sim Y_0$相与作为低3位$F_2\sim F_0$输出，
    两片扩展段$\overline{Y_{\mathrm{EX}}}$相与作为低3位$F_{\mathrm{EX}}$输出，

    \item 8421BCD码优先编码器
    
    10输入4输出。

    $I_0\sim I_7$正常通过非门后从$F_0\sim F_2$输出，
    $I_8$、$I_9$与非后从$F_3$输出，$I_9$和$Y_0$与非后从$F_0$输出。
  \end{itemize}
\end{enumerate}

\subsubsection{译码器}

\begin{enumerate}
  \item 2线至4线译码器
  \begin{lstlisting}[language=Verilog]
  module decode24(a, led);
    input [1:0] a;
    output [3:0] led;
    reg [3:0] led;

    always @(a)
    begin
        case (a)
            2'b00: led = 4'b0001;
            2'b01: led = 4'b0010;
            2'b10: led = 4'b0100;
            2'b11: led = 4'b1000;
        endcase
    end
  endmodule
  \end{lstlisting}
  \item 三线至八线译码器74LS138
  
  $S_\mathrm{A}$为1，$S_\mathrm{B}$、$S_\mathrm{C}$为0时，能正常译码，
  否则输出端全为1，表示无输出。

  \begin{lstlisting}[language=Verilog]
  module decoder3to8(out, in);

      output [7:0] out;
      input [2:0] in;

      assign out = 1'b1 << in;  // 根据输入进行左移，输出'1'有效
  endmodule
  \end{lstlisting}
  \begin{lstlisting}[language=Verilog]
  module decoder3to8(sw, led);
      input [2:0] sw;             // 开关输入信号，作为3-8译码器的输入信号控制LED
      output [7:0] led;
      reg [7:0] led;              // 在always过程中只能对reg型变量赋值

      always @(sw)                // always过程块，括号中sw为敏感变量；
                                  //当sw变化一次，执行一次always中所有语句；否则保持不变
      begin
          case(sw)                // case语句
              3'b000: led = 8'b0111_1111;
              3'b001: led = 8'b1011_1111;
              3'b010: led = 8'b1101_1111;
              3'b011: led = 8'b1110_1111;
              3'b100: led = 8'b1111_0111;
              3'b101: led = 8'b1111_1011;
              3'b110: led = 8'b1111_1101;
              3'b111: led = 8'b1111_1110;
          endcase
      end
  endmodule
  \end{lstlisting}

  功能扩展
  \begin{itemize}
    \item 4线至16线译码器（串行拓展）
    
    片Ⅰ的$S_\mathrm{B}$与片Ⅱ的$S_\mathrm{A}$并接在一起作为最高位$D$的输入端。
    片Ⅰ的$S_\mathrm{C}$与片Ⅱ的$S_\mathrm{B}$连在一起作为使能端。

    \item 6线至64线译码器（并行拓展）
    
    在8片74LS138的基础上添加1片74LS138作为高3位译码器，
    将其输出结果输入到8片74LS138的$S_\mathrm{B}$、$S_\mathrm{C}$端口。

    \item 地址译码器
    
    将地址低3位输入到$A_2\sim A_0$端，
    其他位输入到$S_\mathrm{A}$、$S_\mathrm{B}$、$S_\mathrm{C}$端口，
    保证始终正常工作。

    可以将所有1通过与门，所有0通过或门，接入使能端，即可实现地址译码器。

    \item 数据分配器
    
    将数据$D$输入到$S_\mathrm{B}$、$S_\mathrm{C}$端口。
    
    当$D=1$时，译码器所有端口输出1，数据分配器分配的数据必定是1；
    当$D=0$时，译码器对应端口输出0。

    \item 函数发生器
    
    可以将译码器的输出端看作$2^n$个最小项，可以通过基本逻辑门组合实现任意逻辑函数。
    
    注意：译码器为\textbf{负逻辑输出}！

    \begin{methodbox}[【技巧】最小项间转换]
        $m_0 + m_1 + m_2 = \overline{m_3}$
        
        运用此式可以化简逻辑函数，减少逻辑门数量。
    \end{methodbox}
  \end{itemize}

  \item 二——十进制译码器（4线-10线译码器）
  
  功能扩展
  \begin{itemize}
    \item 3线至8线译码器
    
    $f_8$、$f_9$不用即可。
  \end{itemize}

  \item 七段荧光数码管显示器74LS48
  
  \begin{methodbox}
    共阴极：统一接低电平，输出端口为正逻辑，即1亮0灭；

    共阳极：统一接高电平，输出端口为负逻辑，即0亮1灭；
  \end{methodbox}

  译码器将8421BCD码映射为七段荧光数码管的七个段的点亮情况。

  除去4输入7输出端口外，有如下特殊端口：
  
  \begin{itemize}
    \item $\overline{BI}$：强制灭灯，当$\overline{BI}=1$时，强制全灭，优先级最高；
    \item $\overline{LT}$：试灯，当$\overline{LT}=1$时，强制全亮；
    \item $\overline{RBI}$：灭0输入信号，当$\overline{RBI}=1$时灭0；
    \item $\overline{RBO}$：灭0输入信号，当本数码管显示0时输出0；
  \end{itemize}

  只需在整数部分把高位的$\overline{RBO}$与低位的$\overline{RBI}$相连，
  在小数部分将低位的$\overline{RBO}$与高位的$\overline{RBI}$相连，
  就可以熄灭前后多余的0。
\end{enumerate}

\subsubsection{数据选择器}

\begin{enumerate}
  \item TTL 8选1数据选择器74LS151
  
  地址端$A_2,A_1,A_0$，数据端$D_7\sim D_0$，
  使能端$\overline{ST}$输出端$Y$和$\overline{W}$。
  
  \item 双4选1数据选择器74LS153
  
  地址端$A_1$和$A_0$，数据端$D_3\sim D_0$，
  使能端$\overline{1ST}$和$\overline{2ST}$，输出端$1Y$和$2Y$。

  \begin{lstlisting}[language=Verilog]
  module mux4_to_1(out, i0, i1, i2, i3, s0, s1);

      output out;
      input i0, i1, i2, i3, s0, s1;
      reg out;

      always @(s1 or s0 or i0 or i1 or i2 or i3)
                                  // 用always块建立组合逻辑模型时，要用阻塞赋值
      begin
          case({s1, s0})
              2'b00: out = i0;
              2'b01: out = i1;
              2'b10: out = i2;
              2'b11: out = i3;
              default: out = 1'bx; // 其他情况下输出1位任意值
          endcase
      end

  endmodule
  \end{lstlisting}

  功能扩展

  \begin{itemize}
    \item 8选1数据选择器
    
    将$A_2$与其反变量输入$\overline{1ST}$与$\overline{2ST}$，对输出求或。

    \item 16选1数据选择器
    
    将$A_0$和$A_1$输入第一级电路，16中选4，然后再将结果输入2级电路，用$A_2$和$A_3$进行选择，从4个输出中选1。
  
    \item 并串转换电路
    
    在地址端输入$00\to01\to10\to11\to00\to\cdots$，即可输出$D_0D_1D_2D_3\cdots$。

    \item 函数发生器
    
    若$m\leq n$，则将$2^m$个最小项输入到地址端。

    若$m>n$，则将$2^n$个最小项输入到地址端，剩余的最小项通过逻辑门组合后输入到数据端。
    一般选择$2^n$个最小项中最多的作为数据端输入，剩余的作为地址端输入。

    \begin{methodbox}[Karnaugh图法]
      根据地址端的最小项圈出Karnaugh图中的子Karnaugh图，
      再在子Karnaugh图中圈出最小项，
      最后将圈出的最小项通过逻辑门组合后输入到数据端。
    \end{methodbox}
  \end{itemize}
\end{enumerate}

\subsubsection{数据分配器}

\begin{enumerate}
  \item TTL 双1线至4线数据分配器74LS155
  
  可以视为2到4线译码器添加了使能端与数据输入端的$\overline{D}$与门。

  注意：上下使能端的\textbf{正负逻辑相反}。

  \begin{lstlisting}[language=Verilog]
  module demux4(y0, y1, y2, y3, din, a);

      input din;
      input [1:0] a;
      output reg y0, y1, y2, y3;

      always @(din, a)
      begin
          y0 = 0; y1 = 0; y2 = 0; y3 = 0;
                                  // 用always块建立组合逻辑模型时，要用阻塞赋值
          case(a)
              2'b00: y0 = din;
              2'b01: y1 = din;
              2'b10: y2 = din;
              2'b11: y3 = din;
          endcase
      end

  endmodule
  \end{lstlisting}

  功能扩展
  \begin{itemize}
    \item 8选1数据分配器
    
    将$A_2$与其反变量输入$\overline{1ST}$与$\overline{2ST}$，对输出求或。

    \item 译码器
    
    令$\overline{D}=0$，即可实现译码器功能。
  \end{itemize}
\end{enumerate}

\section{Verilog硬件描述语言（Verilog HDL）}

Verilog HDL既是一种行为描述语言也是一种结构描述语言，即既可以\textbf{用电路的功能描述}，
也可以\textbf{用元器件及其之间的连接}来建立Verilog HDL模型。

Verilog HDL可以描述系统级、算法级、寄存器传输级（RTL）、门级和开关级五种级别的电路。
其中前三种称为行为描述，后两种称为结构描述。

\subsection{数据类型}

\subsubsection{数值}

\begin{enumerate}
  \item 数字
  
  表示：位数+\tt{'}+数制+数值。
  例：\tt{4'b1010}表示4位二进制数1010，\tt{8'hFF}表示8位十六进制数FF。

  数制：\tt{b}表示二进制，\tt{o}表示八进制，\tt{d}表示十进制，\tt{h}表示十六进制。

  特殊：
  \begin{itemize}
    \item 负号要放在最前面，如\tt{-4'b1010}；
    \item 不写位数默认为最长值，不写数制默认为十进制；
    \item \tt{'1}表示全1，\tt{'0}表示全0。
  \end{itemize}
  
  \item 字符串
  
  可以看作特殊的长度为8 bit的二进制数，\tt{"A"}表示8位二进制数01000001。
  字符串的长度为$8\times\text{字符个数}$bit。

  \item 参数\tt{parameter}
\end{enumerate}

\subsubsection{变量}

  \tt{wire}常用于\,\textbf{\tt{assign}语句}的左值；
  
  \tt{reg}常用于\,\textbf{\tt{always}块中\tt{=}或\tt{<=}语句}的左值。

  此外还有\tt{memory}等类型，不多介绍。

\subsection{运算符}

大部分运算符与C语言相同。下面介绍几种不一样的：

\begin{enumerate}
  \item 拼接运算符\tt{\{\}}
  
  直接把所有表达式的值按序拼接。
  
  \item 阻塞赋值和非阻塞赋值
  
  阻塞赋值的符号是\tt{=}，\textbf{只用于组合逻辑电路中}，即\tt{assign}语句或\tt{always @ *}；
  
  非阻塞赋值的符号是\tt{<=}，\textbf{只用于时序逻辑电路中}，即\tt{always @ (posedge clk)}或类似语句。
\end{enumerate}
  
\subsection{语句}

\begin{enumerate}
  \item 赋值

\begin{itemize}
  \item 连续赋值\tt{assign}
  
  用于组合逻辑电路的连续赋值，\textbf{左值必须是\tt{wire}}。
  
  \item \tt{always}块
  
  用于组合逻辑电路或时序逻辑电路的过程赋值，\textbf{左值必须是\tt{reg}}。
  
  \item \tt{initial}块
  
  用于仿真初始化，\textbf{左值必须是\tt{reg}}。
\end{itemize}

  \item 条件

\tt{0}、\tt{x}、\tt{z}均为逻辑0，\tt{1}为逻辑1。

  \item 循环
  
  除\tt{while}、\tt{for}外，还有\tt{repeat}、\tt{forever}。

  \begin{itemize}
    \item \tt{repeat}语句：重复执行指定次数的循环体；

    \item \tt{forever}语句：重复执行循环体，直到仿真结束。
  \end{itemize}

\end{enumerate}
