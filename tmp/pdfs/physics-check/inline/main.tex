\nabla\times\vect{E} &= -\frac{\partial\vect{B}}{\partial t}\\
\nabla\times\vect{B} &= \mu_0\vect{J}
+\mu_0\varepsilon_0\frac{\partial\vect{E}}{\partial t}
\end{aligned}
\right.
\)\par

& \centering\arraybackslash 积分形式

\(\displaystyle
\left\{
\begin{aligned}
\oiint_S \vect{E}\cdot\dif \vect{S}
&= \frac{Q_{\mathrm{in}}}{\varepsilon_0}\\
\oiint_S \vect{B}\cdot\dif \vect{S}
&= 0\\
\oint_C \vect{E}\cdot\dif \vect{l}
&= -\frac{\dif}{\dif t}
\iint_S \vect{B}\cdot\dif \vect{S}\\
\oint_C \vect{B}\cdot\dif \vect{l}
&= \mu_0 I_{\mathrm{in}}
+\mu_0\varepsilon_0\frac{\dif}{\dif t}
\iint_S \vect{E}\cdot\dif \vect{S}
\end{aligned}
\right.
