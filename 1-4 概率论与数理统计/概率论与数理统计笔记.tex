    正态函数线性组合仍然是正态分布，满足
    \[
    C_0+\sum_{i=1}^{n}C_i N_i(\mu_i,\sigma_i^2) \sim N\left(C_0+\sum_{i=1}^{n}C_i \mu_i,\,\sum_{i=1}^{n}C_i^2 \sigma_i^2\right)

  \[
  F_M(x)=\prod_{i=1}^{n}F_i(x)
  \]

  \[
  F_M(x)=[F_X(x)]^n

  \[
  1-F_N(x)=\prod_{i=1}^{n}[1-F_i(x)]

  1-F_N(x)=[1-F_X(x)]^n

\subsection{数字特征}

\subsubsection{期望$E(X)$}

\begin{enumerate}
  \item 计算
  
  离散型

  \[
  E(X) = \sum x_k p_k
  \]

  连续型

  \[
  E(X) = \int_{-\infty}^{+\infty} x f(x)\dd x
  \]

  \item 函数$Y=g(\vect{X})$
  
  离散型

  \[
  E(Y) = \sum g(\vect x_k) p_k  
  \]

  连续型

  \[
  E(Y) = \int_{-\infty}^{+\infty} g(\vect x) f(\vect x)\dd \vect x
  \]

  \begin{methodbox}
  两正交Gauss函数之和的包络服从Rayleigh分布，满足$E(X)=\sigma\sqrt{\dfrac{\uppi}{2}}$。
  \end{methodbox}

  \item 性质
  \begin{enumerate}[label=(\arabic*)]
    \item $E(C)=C$
    
    \item 线性
    
    \[
    E\left(\sum_{i=1}^{n}a_i X_i\right)=\sum_{i=1}^{n}a_i E(X_i)
    \]
    
    注：其线性\textbf{不依赖}于$X_1,\,X_2,\,\cdots,\,X_n$间的独立性。

    \item 乘法性质
    
    若$X_1,\,X_2,\,\cdots,\,X_n$之间是独立的，则有

    \[
    E\left(\prod_{i=1}^{n}X_i\right)=\prod_{i=1}^{n}E(X_i)
    \]

    注：其\textbf{逆命题不成立}，即$E(XY)=E(X)E(Y)$不一定意味着$X$与$Y$独立。

    \item Cauchy-Schwarz不等式
    
    \[
    E[(XY)^2] \leq E(X^2)E(Y^2)
    \]
  \end{enumerate}
\end{enumerate}

\subsubsection{方差$D(X)$}

\begin{enumerate}
  \item 定义
  
  \[
  D(X) = E\left[(X-E(X))^2\right]
  \]

  \begin{methodbox}[标准差$\sigma(X)$]
    为使度量单位相同，定义标准差（协方差）为
    
    \[
    \sigma(X) = \sqrt{D(X)}
    \]
  \end{methodbox}

  对于离散型，有
  \[
  D(X) = \sum \left[x_k - E(X)\right]^2 p_k
  \]

  对于连续型，有
  \[
  D(X) = \int_{-\infty}^{+\infty} [x - E(X)]^2 f(x)\dd x
  \]

  \item 计算公式
  
  \[
  D(X) = E(X^2) - [E(X)]^2
  \]

  \begin{methodbox}[引申公式]
    \[
    E(X^2) \geq [E(X)]^2
    \]

    故，若$E(X^2)=0$，则$E(X)=0$。
  \end{methodbox}

  \item 性质
  \begin{enumerate}[label=(\arabic*)]
    \item $D(C)=0$
  
    这是充要条件。

    \item $D(CX)=C^2 D(X)$
    
    \item $D(X \pm Y)$ = $D(X) + D(Y)$
    
    \[
    D\left(\sum_{i=1}^{n}a_i X_i\right)=\sum_{i=1}^{n}a_i^2 D(X_i)
    \]

    注：$X+Y$与$X-Y$的\textbf{方差相同}。
    \item 若$C \ne E(X)$，则$D(X) < E[(X - C)^2]$
  \end{enumerate}

  \item \russianserif Чебышёв不等式
  
  对于任意$\varepsilon>0$，有

  \[
  P(|X - E(X)| \geq \varepsilon) \leq \frac{D(X)}{\varepsilon^2}
  \]

  \textbf{{\russiansans Чебышёв}不等式是目前估计概率唯一方法！}
\end{enumerate}

\subsubsection{协方差$\mathbf{Cov}$}

\begin{enumerate}
  \item 定义

  \[
  \mathbf{Cov}(\vect X,\vect Y) = \vect E[(\vect X-\vect E(\vect X))(\vect Y-\vect E(\vect Y))]
  \]

  \item 重要计算公式
  \[
  \operatorname{Cov}(X,Y) = E(XY) - E(X)E(Y)
  \]
  \[
  \operatorname{Cov}(X, Y) = \frac{1}{2} (D(X+Y) - D(X) - D(Y))
  \]

  \item 性质
  \begin{enumerate}[label=(\arabic*)]
    \item $\operatorname{Cov}(X,X) = D(X)$
    \item $\operatorname{Cov}(X,C) = 0$
    \item $\operatorname{Cov}(Y,X) = \operatorname{Cov}(X,Y)$
    \item $\operatorname{Cov}(aX,bY) = ab \operatorname{Cov}(X,Y)$
    \item $\operatorname{Cov}(X+Y,Z) = \operatorname{Cov}(X,Z) + \operatorname{Cov}(Y,Z)$
  \end{enumerate}

  \item 意义
  
  $X\text{和}Y\text{独立} \Longrightarrow \Cov (X, Y)=0$，其逆命题不成立

  \item 相关性
  
  若$\Cov (X, Y)>0$，则$X$与$Y$正相关；
  
  若$\Cov (X, Y)<0$，则$X$与$Y$负相关；
  
  若$\Cov (X, Y)=0$，则$X$与$Y$不相关。
\end{enumerate}

\subsubsection{相关系数$\rho$}
\begin{enumerate}
  \item 定义 

  作数据标准化
  \[
  X^{\ast} = \frac{X-\vect E(X)}{\sigma (X)}
  \]

  使得$E(X^{\ast})=0$，$D(X^{\ast})=1$，定义相关系数为

  \[
  \rho_{XY} = \Cov(X^{\ast}, Y^{\ast}) = \frac{\Cov(X, Y)}{\sigma(X)\sigma(Y)}
  \]
  
  \item 性质
  \begin{enumerate}[label=(\arabic*)]
    \item $|\rho_{XY}| \leq 1$
    \item $|\rho_{XY}|=1$当且仅当$X$与$Y$线性相关。
    \item $|\rho_{XY}|=0$当且仅当$X$与$Y$不线性相关。
  \end{enumerate}

  但二维正态随机变量$(X,Y)$的不相关与独立等价。
\end{enumerate}

    \item $\vect C$是对称矩阵。
  \end{enumerate}

\end{enumerate}

\subsection{大数定律}

\begin{enumerate}
  \item Bernoulli大数定律
  
  \[
  \lim_{n\to\infty} P\left\{\left|\frac{n_A}{n} - p\right| \geq \varepsilon\right\} = 0
  \]

  这表明当$n$足够大时，频率稳定于概率，即$n\to\infty,\quad\dfrac{n_A}{n}\to p$。

  \item {\russianserif Чебышёв}大数定律
  
  \[
  \lim_{n\to\infty} P\left\{\left|\frac{1}{n}\sum_{i=1}^{n}X_i - \mu\right| \geq \varepsilon\right\} = 0
  \]

  这表明当$n$足够大时，均值稳定于期望，
  即$n\to\infty,\quad X \to \mu$。

  \item 弱大数定律
  
  又称大数定律、{\russianserif Хинчин}大数定律。

  \[
  \lim_{n\to\infty} P\left\{\left|\frac{1}{n}\sum_{i=1}^{n}X_i - \mu\right| \geq \varepsilon\right\} = 0
  \]
  
  \item 强大数定律
  
  \[
  \lim_{n\to\infty} P\left\{\left|\frac{1}{n}\sum_{i=1}^{n}X_i - 0\right| = 0\right\} = 1
  \]
\end{enumerate}

\begin{methodbox}[Monte Carlo方法]
  Monte Carlo方法是利用计算机模拟随机事件的发生过程，进而求解概率问题的一种数值计算方法。
  
\end{methodbox}
