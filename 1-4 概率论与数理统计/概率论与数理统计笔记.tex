\subsubsection{随机变量函数}
  一般随机变量函数

  \begin{enumerate}

  \item 和函数$Z=X+Y$的分布
  
  \[
  f(z) = f(x) \ast f(y)
  \]

  故有

  \[
  f(z) = \int_{-\infty}^{\infty} f(x)f(z-x)\dd x = \int_{-\infty}^{\infty} f(z-y)f(y)\dd y
  \]

  二项分布、Poisson分布、正态分布、$\Gamma$分布等分布均具有可加性，即：
  \[
  b(n_1,\,p)+b(n_2,\,p)=b(n_1+n_2,\,p)
  \]
  \[
  \pi(\lambda_1)+\pi(\lambda_2)=\pi(\lambda_1+\lambda_2)
  \]
  \[
  N(\mu_1,\sigma_1^2)+N(\mu_2,\sigma_2^2)=N(\mu_1+\mu_2,\sigma_1^2+\sigma_2^2)
  \]

  \begin{methodbox}[正态分布线性组合]
    正态函数线性组合仍然是正态分布，满足
    \[
    C_0+\sum_{i=1}^{n}C_i N_i(\mu_i,\sigma_i^2) \sim N\left(C_0+\sum_{i=1}^{n}C_i \mu_i,\,\sum_{i=1}^{n}C_i^2 \sigma_i^2\right)
    \]
  \end{methodbox}

  \item $M=\max\left\{X,\,Y\right\}$
  
  \[
  F_M(x)=\prod_{i=1}^{n}F_i(x)
  \]
  
  特殊地，当$X_1,\,X_2,\,\cdots,\,X_n$相互独立且同分布时，有
  
  \[
  F_M(x)=[F_X(x)]^n
  \]

  \item $N=\min\left\{X,\,Y\right\}$
  
  \[
  1-F_N(x)=\prod_{i=1}^{n}[1-F_i(x)]
  \]
  
  特殊地，当$X_1,\,X_2,\,\cdots,\,X_n$相互独立且同分布时，有
  
  \[
  1-F_N(x)=[1-F_X(x)]^n
  \]

\end{enumerate}

\begin{methodbox}[$\chi^2$分布]
  形如$Y=\sum_{i=1}^{n}X_i^2,\,X_i\sim N(0,\,1)$的分布称为$\chi^2$分布，记为$Y\sim \chi^2(n)$。
\end{methodbox}

\begin{methodbox}[Gauss积分公式]
  \[
  \int_{-\infty}^{+\infty} \ee^{-x^2}\dd x=\sqrt{\uppi}
  \]
  
  对于一般函数$f(x)=\ee^{-ax^2+bx+c}$，有

  \[
  \int_{-\infty}^{+\infty} \ee^{-ax^2+bx+c}\dd x=\sqrt{\frac{\uppi}{a}}\ee^{\frac{b^2}{4a}+c}
  \]
\end{methodbox}
